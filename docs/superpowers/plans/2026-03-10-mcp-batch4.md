```python
from db import get_readonly_connection


def _paper_trades_freshness(conn=None) -> float | None:
    """Get most recent paper_trades closed_at as epoch timestamp."""
    close_conn = False
    if conn is None:
        conn = get_readonly_connection()
        close_conn = True
    try:
        row = conn.execute(
            "SELECT MAX(closed_at) as mca FROM paper_trades WHERE status != 'open'"
        ).fetchone()
        if row and row["mca"]:
            from datetime import datetime, timezone
            dt_obj = datetime.strptime(row["mca"], "%Y-%m-%d %H:%M:%S")
            return dt_obj.replace(tzinfo=timezone.utc).timestamp()
    except Exception:
        pass
    finally:
        if close_conn:
            conn.close()
    return None
```

```python
"""Shared strategy validation and mutation helpers.

Used by both routes/api_strategies.py and mcp_tools/write_tools.py.
"""
from __future__ import annotations

import json
import re
import sqlite3

from strategies.registry import list_strategy_plugins

WALLET_RE = re.compile(r"^0x[a-fA-F0-9]{40}$")

VALID_STRATEGY_TYPES = {
    "mirror_all",
    "selective",
    "signal_only",
    "fade",
    *(p.plugin_id for p in list_strategy_plugins()),
}

VALID_STRATEGY_STATUSES = {"active", "paused", "stopped"}


def normalize_wallet(wallet: str | None) -> str:
    """Normalize wallet to lowercase stripped string."""
    return str(wallet or "").strip().lower()


def validate_wallet(wallet: str | None) -> str:
    """Normalize and validate wallet. Returns normalized wallet or raises ValueError."""
    w = normalize_wallet(wallet)
    if not WALLET_RE.fullmatch(w):
        raise ValueError("wallet must be a valid 0x address")
    return w


def validate_strategy_type(strategy_type: str | None) -> str:
    """Normalize and validate strategy_type. Raises ValueError if invalid."""
    st = str(strategy_type or "").strip().lower()
    if st not in VALID_STRATEGY_TYPES:
        raise ValueError(
            f"strategy_type must be one of: {', '.join(sorted(VALID_STRATEGY_TYPES))}"
        )
    return st


def validate_parameters(parameters) -> dict:
    """Ensure parameters is a dict. Parses JSON strings. Returns {} on failure."""
    if isinstance(parameters, dict):
        return parameters
    if isinstance(parameters, str) and parameters:
        try:
            parsed = json.loads(parameters)
            if isinstance(parsed, dict):
                return parsed
        except Exception:
            pass
    return {}


def create_strategy(
    conn: sqlite3.Connection,
    *,
    wallet: str,
    label: str,
    strategy_type: str,
    parameters: dict,
) -> dict:
    """Validate inputs, create strategy in paused state, return row as dict.

    Raises ValueError on invalid inputs.
    """
    wallet = validate_wallet(wallet)
    strategy_type = validate_strategy_type(strategy_type)
    parameters = validate_parameters(parameters)
    label = str(label or "").strip()

    from db import insert_strategy

    strategy_id = insert_strategy(
        conn,
        wallet=wallet,
        label=label,
        strategy_type=strategy_type,
        parameters=parameters,
        status="paused",
    )

    row = conn.execute(
        "SELECT * FROM strategies WHERE id = ?", (strategy_id,)
    ).fetchone()
    return dict(row) if row else {"id": strategy_id, "status": "paused"}


def activate_strategy(
    conn: sqlite3.Connection,
    strategy_id: int,
) -> dict:
    """Activate a paused strategy. Raises ValueError if not paused or not found."""
    row = conn.execute(
        "SELECT * FROM strategies WHERE id = ?", (strategy_id,)
    ).fetchone()
    if not row:
        raise ValueError(f"strategy {strategy_id} not found")
    if row["status"] != "paused":
        raise ValueError(
            f"strategy {strategy_id} status is '{row['status']}', must be 'paused' to activate"
        )

    from db import update_strategy_status

    update_strategy_status(conn, strategy_id, "active")

    row = conn.execute(
        "SELECT * FROM strategies WHERE id = ?", (strategy_id,)
    ).fetchone()
    return dict(row) if row else {"id": strategy_id, "status": "active"}
```

```python
"""MCP write tools for Sniper — strategy creation and activation."""
from __future__ import annotations

import json
import logging

from db import get_connection, insert_mcp_audit_log
from mcp_tools.schemas import make_envelope
from strategy_service import create_strategy, activate_strategy

logger = logging.getLogger(__name__)


def create_paper_strategy_impl(
    *,
    wallet: str,
    strategy_type: str,
    label: str = "",
    parameters: str = "{}",
    conn=None,
    db_path: str | None = None,
) -> dict:
    """Internal implementation for create_paper_strategy tool."""
    close_conn = False
    if conn is None:
        from db import get_db_path
        conn = get_connection(db_path or get_db_path())
        close_conn = True

    params_dict = {}
    if isinstance(parameters, str):
        try:
            params_dict = json.loads(parameters)
        except Exception:
            params_dict = {}
    elif isinstance(parameters, dict):
        params_dict = parameters

    audit_params = json.dumps({
        "wallet": wallet, "strategy_type": strategy_type,
        "label": label, "parameters": params_dict,
    })

    try:
        result = create_strategy(
            conn,
            wallet=wallet,
            label=label,
            strategy_type=strategy_type,
            parameters=params_dict,
        )
        insert_mcp_audit_log(
            conn, tool_name="create_paper_strategy",
            params=audit_params, result=json.dumps(result, default=str),
            success=True,
        )
        return make_envelope(result, data_source="strategies")
    except ValueError as e:
        insert_mcp_audit_log(
            conn, tool_name="create_paper_strategy",
            params=audit_params, result=None,
            success=False, error_msg=str(e),
        )
        return make_envelope({"error": str(e)}, data_source="strategies")
    finally:
        if close_conn:
            conn.close()


def activate_paper_strategy_impl(
    *,
    strategy_id: int,
    conn=None,
    db_path: str | None = None,
) -> dict:
    """Internal implementation for activate_paper_strategy tool."""
    close_conn = False
    if conn is None:
        from db import get_db_path
        conn = get_connection(db_path or get_db_path())
        close_conn = True

    audit_params = json.dumps({"strategy_id": strategy_id})

    try:
        result = activate_strategy(conn, strategy_id)
        insert_mcp_audit_log(
            conn, tool_name="activate_paper_strategy",
            params=audit_params, result=json.dumps(result, default=str),
            success=True,
        )
        return make_envelope(result, data_source="strategies")
    except ValueError as e:
        insert_mcp_audit_log(
            conn, tool_name="activate_paper_strategy",
            params=audit_params, result=None,
            success=False, error_msg=str(e),
        )
        return make_envelope({"error": str(e)}, data_source="strategies")
    finally:
        if close_conn:
            conn.close()


def register_write_tools(mcp_server):
    """Register write tools with the MCP server."""

    @mcp_server.tool()
    def create_paper_strategy(
        wallet: str,
        strategy_type: str,
        label: str = "",
        parameters: str = "{}",
    ) -> dict:
        """Create a new paper trading strategy in paused state. Wallet must be a valid 0x address. strategy_type must be a registered plugin (e.g. 'insider_copy_v1', 'fade_v1'). Parameters is a JSON string of plugin config."""
        return create_paper_strategy_impl(
            wallet=wallet,
            strategy_type=strategy_type,
            label=label,
            parameters=parameters,
        )

    @mcp_server.tool()
    def activate_paper_strategy(strategy_id: int) -> dict:
        """Activate a paused paper strategy. Only works if the strategy's current status is 'paused'. Returns error if already active or not found."""
        return activate_paper_strategy_impl(strategy_id=strategy_id)
```
